## 1. 架构概览

```
┌──────────────────────────────────────────────────────┐
│              智能农业 Agent (Streamlit UI)              │
│                                                      │
│  注册设备 → 选择驱动 → 配置连接参数 → 保存             │
│     │          │                                     │
│     │    ┌─────┼──────────────────────────┐          │
│     ▼    │     ▼                          ▼          │
│  Simulator  MQTT Driver    HTTP Driver    Modbus      │
│  (本地内存)  (MQTT Broker)  (REST API)    (串口/TCP)  │
│     │          │               │             │        │
│     ▼          ▼               ▼             ▼        │
│  虚拟设备   真实MQTT设备    真实HTTP设备   真实PLC/传感器 │
└──────────────────────────────────────────────────────┘
```

**核心原理：** Agent 下发统一的 `DeviceCommand` → 驱动层翻译成协议特定的消息 → 设备执行。

## 3. MQTT 设备连接

### 3.1 协议说明

```
Agent ──发布指令──▶ MQTT Broker ──订阅──▶ ESP32/ESP8266/树莓派 ──▶ 继电器/传感器
Agent ◀──订阅状态── MQTT Broker ◀──发布── ESP32/ESP8266/树莓派 ◀── 传感器读数
```

**消息格式（Agent → 设备）：**
```json
{
  "command": "start",
  "params": {"duration": 30},
  "timestamp": "2026-06-14T10:00:00",
  "device_id": "greenhouse_pump_01"
}
```

**消息格式（设备 → Agent，状态上报）：**
```json
{
  "power": true,
  "status": "running",
  "flow_rate": 2.5,
  "total_water_liters": 150,
  "timestamp": "2026-06-14T10:00:05"
}
```

### 3.3 ESP32 设备端示例代码（Arduino / PlatformIO）

```cpp
/*
 * ESP32 智能灌溉控制器
 * 功能：订阅 MQTT 控制主题，控制继电器（水泵），上报状态
 *
 * 接线：
 *   ESP32 GPIO26 → 继电器模块 IN → 继电器 NO → 水泵 → 电源
 *
 * 依赖库 (PlatformIO):
 *   pubsubclient >= 2.8
 *   ArduinoJson >= 6.0
 */

#include <WiFi.h>
#include <PubSubClient.h>
#include <ArduinoJson.h>

// ===== 配置 =====
const char* WIFI_SSID = "你的WiFi名";
const char* WIFI_PASSWORD = "你的WiFi密码";
const char* MQTT_BROKER = "192.168.1.100";  // MQTT Broker IP
const int   MQTT_PORT = 1883;
const char* DEVICE_ID = "greenhouse_pump_01";
const char* CONTROL_TOPIC = "greenhouse/pump/control";
const char* STATE_TOPIC = "greenhouse/pump/state";

// 继电器引脚
const int RELAY_PIN = 26;  // GPIO26 控制继电器

// ===== 全局状态 =====
WiFiClient espClient;
PubSubClient client(espClient);
bool pump_running = false;
unsigned long pump_start_time = 0;
unsigned long pump_duration_sec = 0;  // 计划运行时长（秒）
float total_water_liters = 0;         // 累计水量
unsigned long last_state_report = 0;

// ===== WiFi 连接 =====
void setup_wifi() {
  Serial.print("连接 WiFi");
  WiFi.begin(WIFI_SSID, WIFI_PASSWORD);
  int attempts = 0;
  while (WiFi.status() != WL_CONNECTED && attempts < 30) {
    delay(500);
    Serial.print(".");
    attempts++;
  }
  if (WiFi.status() == WL_CONNECTED) {
    Serial.println("\nWiFi 已连接, IP: " + WiFi.localIP().toString());
  } else {
    Serial.println("\nWiFi 连接失败，重启中...");
    ESP.restart();
  }
}

// ===== MQTT 回调：收到控制指令 =====
void mqtt_callback(char* topic, byte* payload, unsigned int length) {
  // 解析 JSON 指令
  StaticJsonDocument<256> doc;
  DeserializationError err = deserializeJson(doc, payload, length);
  if (err) {
    Serial.println("JSON 解析失败: " + String(err.c_str()));
    return;
  }

  const char* command = doc["command"];
  int duration = doc["params"]["duration"] | 30;  // 默认30分钟

  Serial.printf("收到指令: command=%s, duration=%d min\n", command, duration);

  if (strcmp(command, "start") == 0) {
    // 启动水泵
    digitalWrite(RELAY_PIN, HIGH);  // 继电器闭合
    pump_running = true;
    pump_start_time = millis();
    pump_duration_sec = duration * 60;  // 转换为秒
    Serial.printf("水泵已启动，计划运行 %d 分钟\n", duration);

    // 上报状态
    report_state();

  } else if (strcmp(command, "stop") == 0) {
    // 停止水泵
    digitalWrite(RELAY_PIN, LOW);
    pump_running = false;
    // 累计水量（假设流量 2 L/min）
    unsigned long elapsed_min = ((millis() - pump_start_time) / 60000);
    total_water_liters += elapsed_min * 2.0;
    pump_duration_sec = 0;
    Serial.println("水泵已停止");
    report_state();
  }
}

// ===== 上报状态到 MQTT =====
void report_state() {
  StaticJsonDocument<256> doc;
  doc["device_id"] = DEVICE_ID;
  doc["power"] = pump_running;
  doc["status"] = pump_running ? "running" : "idle";
  doc["flow_rate"] = pump_running ? 2.0 : 0.0;     // L/min
  doc["total_water_liters"] = total_water_liters;
  doc["wifi_signal"] = WiFi.RSSI();
  doc["uptime_sec"] = millis() / 1000;
  doc["timestamp"] = millis();

  char buffer[256];
  serializeJson(doc, buffer);
  client.publish(STATE_TOPIC, buffer);
  Serial.println("状态已上报: " + String(buffer));
}

// ===== MQTT 重连 =====
void reconnect_mqtt() {
  while (!client.connected()) {
    Serial.print("连接 MQTT...");
    String clientId = String(DEVICE_ID) + "_" + String(random(0xffff), HEX);
    if (client.connect(clientId.c_str())) {
      Serial.println("成功");
      // 订阅控制主题
      client.subscribe(CONTROL_TOPIC);
      Serial.println("已订阅: " + String(CONTROL_TOPIC));
      // 上报初始状态
      report_state();
    } else {
      Serial.printf("失败, rc=%d, 5秒后重试\n", client.state());
      delay(5000);
    }
  }
}

// ===== 初始化 =====
void setup() {
  Serial.begin(115200);
  pinMode(RELAY_PIN, OUTPUT);
  digitalWrite(RELAY_PIN, LOW);  // 初始：继电器断开

  setup_wifi();
  client.setServer(MQTT_BROKER, MQTT_PORT);
  client.setCallback(mqtt_callback);
  client.setKeepAlive(60);
}

// ===== 主循环 =====
void loop() {
  if (!client.connected()) {
    reconnect_mqtt();
  }
  client.loop();

  // 检查是否到时间停止水泵
  if (pump_running && pump_duration_sec > 0) {
    unsigned long elapsed = (millis() - pump_start_time) / 1000;
    if (elapsed >= pump_duration_sec) {
      digitalWrite(RELAY_PIN, LOW);
      pump_running = false;
      total_water_liters += (pump_duration_sec / 60) * 2.0;
      pump_duration_sec = 0;
      Serial.println("定时结束，水泵已自动停止");
      report_state();
    }
  }

  // 每 30 秒上报一次状态
  if (millis() - last_state_report > 30000) {
    report_state();
    last_state_report = millis();
  }
}
```

### 3.4 Python 设备端示例（树莓派 / Jetson Nano）

```python
"""
树莓派 MQTT 设备端 — 控制 GPIO 继电器

依赖:
    pip install paho-mqtt RPi.GPIO

接线:
    GPIO17 → 继电器模块 IN
"""

import json
import time
import signal
import sys
import paho.mqtt.client as mqtt

# ===== 配置 =====
MQTT_BROKER = "192.168.1.100"
MQTT_PORT = 1883
DEVICE_ID = "raspberry_pump_01"
CONTROL_TOPIC = "greenhouse/pump/control"
STATE_TOPIC = "greenhouse/pump/state"

# GPIO 设置
try:
    import RPi.GPIO as GPIO
    GPIO.setmode(GPIO.BCM)
    RELAY_PIN = 17
    GPIO.setup(RELAY_PIN, GPIO.OUT)
    GPIO.output(RELAY_PIN, GPIO.LOW)
    HAS_GPIO = True
except (ImportError, RuntimeError):
    HAS_GPIO = False
    RELAY_PIN = None
    print("⚠️  RPi.GPIO 不可用，进入模拟模式")

# ===== 全局状态 =====
pump_running = False
pump_start_time = 0
pump_duration_sec = 0


def start_pump(duration_minutes: int = 30):
    """启动水泵"""
    global pump_running, pump_start_time, pump_duration_sec
    if HAS_GPIO:
        GPIO.output(RELAY_PIN, GPIO.HIGH)
    pump_running = True
    pump_start_time = time.time()
    pump_duration_sec = duration_minutes * 60
    print(f"💧 水泵启动，计划 {duration_minutes} 分钟")
    report_state()


def stop_pump():
    """停止水泵"""
    global pump_running, pump_duration_sec
    if HAS_GPIO:
        GPIO.output(RELAY_PIN, GPIO.LOW)
    pump_running = False
    pump_duration_sec = 0
    print("⏹️  水泵停止")
    report_state()


def report_state():
    """上报当前状态到 MQTT"""
    state = {
        "device_id": DEVICE_ID,
        "power": pump_running,
        "status": "running" if pump_running else "idle",
        "uptime_sec": int(time.time() - pump_start_time) if pump_running else 0,
        "timestamp": time.time(),
    }
    client.publish(STATE_TOPIC, json.dumps(state))
    print(f"📡 状态已上报: {state}")


# ===== MQTT 回调 =====
def on_connect(client, userdata, flags, rc):
    print(f"✅ MQTT 已连接 (rc={rc})")
    client.subscribe(CONTROL_TOPIC)
    print(f"📡 已订阅: {CONTROL_TOPIC}")
    report_state()


def on_message(client, userdata, msg):
    """收到控制指令"""
    try:
        payload = json.loads(msg.payload.decode("utf-8"))
        command = payload.get("command")
        params = payload.get("params", {})

        print(f"📩 收到指令: {command}, 参数: {params}")

        if command == "start":
            duration = params.get("duration", 30)
            start_pump(duration)
        elif command == "stop":
            stop_pump()

    except json.JSONDecodeError as e:
        print(f"❌ JSON 解析失败: {e}")
    except Exception as e:
        print(f"❌ 处理异常: {e}")


def on_disconnect(client, userdata, rc):
    print(f"⚠️  MQTT 断开 (rc={rc})，自动重连...")


def cleanup(signum, frame):
    """优雅退出"""
    print("\n🛑 正在退出...")
    if pump_running:
        stop_pump()
    client.disconnect()
    if HAS_GPIO:
        GPIO.cleanup()
    sys.exit(0)


# ===== 主程序 =====
if __name__ == "__main__":
    signal.signal(signal.SIGINT, cleanup)
    signal.signal(signal.SIGTERM, cleanup)

    client = mqtt.Client(client_id=DEVICE_ID)
    client.on_connect = on_connect
    client.on_message = on_message
    client.on_disconnect = on_disconnect

    client.connect(MQTT_BROKER, MQTT_PORT, keepalive=60)
    client.loop_start()

    print(f"🚀 设备 {DEVICE_ID} 已启动，等待指令...")
    print(f"   控制主题: {CONTROL_TOPIC}")
    print(f"   状态主题: {STATE_TOPIC}")

    try:
        while True:
            # 检查定时停止
            if pump_running and pump_duration_sec > 0:
                elapsed = time.time() - pump_start_time
                if elapsed >= pump_duration_sec:
                    print("⏰ 定时结束")
                    stop_pump()

            time.sleep(1)

    except KeyboardInterrupt:
        cleanup(None, None)
```

## 4. HTTP REST 设备连接

### 4.1 协议说明

```
Agent ──POST /command──▶ HTTP 设备 (树莓派/ESPHome/Tasmota)
Agent ◀──GET  /state───  HTTP 设备
```

**设备必须实现两个端点：**

| 方法 | 路径 | 说明 |
|------|------|------|
| `POST` | `/command` | 接收控制指令 |
| `GET` | `/state` | 返回当前状态 |

### 4.2 Python Flask 设备端示例（树莓派）

```python
"""
Flask HTTP 设备控制器 — 适合树莓派、NanoPi 等 Linux 单板机

依赖:
    pip install flask

启动:
    python http_device.py
    # 默认监听 http://0.0.0.0:8080
"""

import json
import time
import subprocess
from flask import Flask, request, jsonify

app = Flask(__name__)

# ===== 配置 =====
DEVICE_ID = "smart_plug_01"
DEVICE_NAME = "智能插座#1"

# ===== 全局状态 =====
device_state = {
    "power": False,
    "status": "idle",
    "temperature": 25.0,   # 可从传感器读取
    "uptime_sec": 0,
    "last_command": None,
}


# ── 控制指令接口 ────────────────────────────
@app.route("/command", methods=["POST"])
def handle_command():
    """接收来自 Agent 的控制指令"""
    try:
        data = request.get_json()
        command = data.get("command")
        params = data.get("params", {})

        print(f"📩 收到指令: {command}, 参数: {params}")

        if command == "start":
            device_state["power"] = True
            device_state["status"] = "running"
            device_state["last_command"] = command
            device_state["uptime_sec"] = 0

            # 实际控制硬件 —— 这里用 GPIO 或系统命令
            # 例如通过 GPIO 控制继电器：
            # GPIO.output(RELAY_PIN, GPIO.HIGH)
            # 例如通过串口发送命令：
            # serial.write(b"RELAY_ON\n")
            # 例如执行系统命令：
            # subprocess.run(["tdtool", "--on", "1"])

            print(f"✅ 设备已启动 (参数: {params})")

        elif command == "stop":
            device_state["power"] = False
            device_state["status"] = "idle"
            device_state["last_command"] = command

            # GPIO.output(RELAY_PIN, GPIO.LOW)
            print("⏹️  设备已停止")

        elif command == "set_param":
            # 设置参数（如温度阈值、亮度等）
            for key, val in params.items():
                if key in device_state:
                    device_state[key] = val
            print(f"⚙️  参数已更新: {params}")

        return jsonify({
            "success": True,
            "message": f"指令 '{command}' 已执行",
            "device_id": DEVICE_ID,
        })

    except Exception as e:
        return jsonify({
            "success": False,
            "message": str(e),
        }), 500


# ── 状态查询接口 ────────────────────────────
@app.route("/state", methods=["GET"])
def get_state():
    """返回设备当前状态"""
    device_state["uptime_sec"] += 1  # 模拟运行时间

    # 可选：从真实传感器读取
    # device_state["temperature"] = read_temperature_sensor()

    return jsonify(device_state)


# ── 健康检查 ────────────────────────────────
@app.route("/", methods=["GET"])
def health():
    return jsonify({
        "device_id": DEVICE_ID,
        "name": DEVICE_NAME,
        "status": "online",
        "endpoints": {
            "command": "POST /command",
            "state": "GET /state",
        },
    })


if __name__ == "__main__":
    print(f"🚀 HTTP 设备控制器已启动")
    print(f"   设备ID: {DEVICE_ID}")
    print(f"   地址: http://0.0.0.0:8080")
    print(f"   控制端点: POST http://<IP>:8080/command")
    print(f"   状态端点: GET  http://<IP>:8080/state")
    app.run(host="0.0.0.0", port=8080, debug=False)
```

### 4.3 ESPHome / Tasmota 设备适配

**Tasmota 智能插座（已有固件）：**

Tasmota 默认提供 HTTP API，无需额外编程。只需在 Agent 中配置：

| 字段 | 值 |
|------|-----|
| 设备ID | `tasmota_plug_01` |
| 驱动类型 | HTTP |
| Base URL | `http://192.168.1.101` |

然后需要一个简单的中间层翻译 Tasmota 的 API 格式：

```python
"""
Tasmota 适配器 — 将 Agent 的 HTTP 指令翻译为 Tasmota API 调用

把这个脚本和主 Flask 应用部署在一起，或者单独运行在树莓派上。
"""

from flask import Flask, request, jsonify
import requests

app = Flask(__name__)

TASMOTA_IP = "192.168.1.101"  # Tasmota 设备 IP


@app.route("/command", methods=["POST"])
def handle_command():
    data = request.get_json()
    command = data.get("command")

    if command == "start":
        # Tasmota: 打开继电器
        resp = requests.get(f"http://{TASMOTA_IP}/cm?cmnd=Power%20On", timeout=5)
        return jsonify({"success": True, "message": "Tasmota 已开机"})

    elif command == "stop":
        # Tasmota: 关闭继电器
        resp = requests.get(f"http://{TASMOTA_IP}/cm?cmnd=Power%20Off", timeout=5)
        return jsonify({"success": True, "message": "Tasmota 已关机"})

    return jsonify({"success": False, "message": f"未知指令: {command}"})


@app.route("/state", methods=["GET"])
def get_state():
    # 读取 Tasmota 状态
    try:
        resp = requests.get(f"http://{TASMOTA_IP}/cm?cmnd=Status%208", timeout=3)
        data = resp.json()
        power_state = data.get("StatusSNS", {}).get("ENERGY", {}).get("Power", 0)
        return jsonify({
            "power": power_state > 0,
            "status": "running" if power_state > 0 else "idle",
            "power_watts": power_state,
        })
    except Exception:
        return jsonify({"power": False, "status": "unknown"})


if __name__ == "__main__":
    print("🚀 Tasmota 适配器已启动")
    app.run(host="0.0.0.0", port=8080)
```

### 4.4 在 Agent 中注册 HTTP 设备

| 字段 | 示例值 |
|------|--------|
| 设备ID | `smart_plug_01` |

## 5. Modbus 设备连接

### 5.1 协议说明

Modbus 是工业自动化标准协议，用于 PLC、变频器、传感器等设备。

```
Agent ──Modbus RTU/TCP──▶ USB/RS-485 转换器 ──▶ PLC / 变频器 / 传感器
```

### 5.2 环境准备

```bash
pip install pymodbus
```

### 5.3 Modbus RTU 示例（USB 转 RS-485）

```python
"""
Modbus RTU 设备端模拟器 — 用于测试

启动后模拟一个 Modbus RTU 从设备，可以通过 Agent 的 Modbus 驱动控制。

依赖:
    pip install pymodbus
"""

from pymodbus.server import StartSerialServer
from pymodbus.device import ModbusDeviceIdentification
from pymodbus.datastore import ModbusSequentialDataBlock, ModbusSlaveContext, ModbusServerContext
from pymodbus.transaction import ModbusRtuFramer
import logging

# 配置日志
logging.basicConfig(level=logging.INFO)
logger = logging.getLogger(__name__)

# ===== 配置 =====
SERIAL_PORT = "/dev/ttyUSB0"  # Linux
# SERIAL_PORT = "COM3"        # Windows
BAUDRATE = 9600
SLAVE_ID = 1

# ===== 模拟数据存储 =====
# 保持寄存器 (Holding Registers): 可读写
#   地址 0: 电源开关 (0=关, 1=开)
#   地址 1: 运行状态 (0=空闲, 1=运行中)
#   地址 2: 设定值 (如转速/温度)
#   地址 3: 当前值 (传感器读数)
store = ModbusSlaveContext(
    di=ModbusSequentialDataBlock(0, [0]*100),   # 离散输入
    co=ModbusSequentialDataBlock(0, [0]*100),   # 线圈
    hr=ModbusSequentialDataBlock(0, [0]*100),   # 保持寄存器 ★
    ir=ModbusSequentialDataBlock(0, [0]*100),   # 输入寄存器
)

# 初始化保持寄存器
store.setValues(3, 0, [0, 0, 800, 0])  # 电源关, 空闲, 转速800, 当前值0

context = ModbusServerContext(slaves=store, single=True)


def main():
    """启动 Modbus RTU 从设备模拟器"""
    logger.info("🚀 Modbus RTU 模拟从设备启动中...")
    logger.info(f"   串口: {SERIAL_PORT}")
    logger.info(f"   波特率: {BAUDRATE}")
    logger.info(f"   从站地址: {SLAVE_ID}")
    logger.info(f"   寄存器说明:")
    logger.info(f"     HR[0] = 电源开关 (写 1 启动, 写 0 停止)")
    logger.info(f"     HR[1] = 运行状态 (只读)")
    logger.info(f"     HR[2] = 设定值 (可读写)")
    logger.info(f"     HR[3] = 当前值 (只读，模拟传感器)")

    StartSerialServer(
        context=context,
        framer=ModbusRtuFramer,
        port=SERIAL_PORT,
        baudrate=BAUDRATE,
        timeout=1,
    )


if __name__ == "__main__":
    main()
```

### 5.4 真实 PLC 设备接线

```
电脑 USB → USB-to-RS485 转换器 → A+/B- 双绞线 → PLC RS-485 端口
```

| 转换器引脚 | PLC 引脚 | 说明 |
|-----------|---------|------|
| A+ (或 D+) | A+ / 485+ | 数据正 |
| B- (或 D-) | B- / 485- | 数据负 |
| GND | GND | 共地（可选但推荐） |

|------|--------|

## 6. 设备端实现规范

无论使用哪种协议，设备端只需遵循以下约定即可无缝接入：

### 6.1 指令格式（Agent → 设备）

```json
{
  "command": "start|stop|set_param",
  "params": {
    "duration": 30,          // 灌溉/通风时长（分钟）
    "amount_kg": 5,          // 施肥量（kg）
    "target_temp": 22,       // 目标温度（°C）
    "brightness_percent": 80, // 亮度百分比
    "flow_rate": "medium"    // 流量等级
  },
  "timestamp": "2026-06-14T10:00:00",
  "device_id": "your_device_id"
}
```

### 6.2 状态格式（设备 → Agent）

```json
{
  "power": true|false,
  "status": "running|idle|error",
  "temperature": 22.5,       // 可选：温度
  "humidity": 65.0,          // 可选：湿度
  "soil_moisture": 28.0,     // 可选：土壤湿度
  "flow_rate": 2.0,          // 可选：流量
  "uptime_sec": 120,         // 可选：运行时长
  "error_message": "",       // 可选：错误信息
  "timestamp": 1234567890
}
```

### 6.3 安全要求

- 设备端应内置**最大运行时间保护**（如水泵最多连续运行 2 小时自动停止）
- 设备离线时应**自动切换到安全状态**（关闭继电器等）
- 建议设备端也实现**物理急停按钮**

## 8. 故障排查

### MQTT 设备不可达

```bash
# 1. 检查 Broker 是否运行
systemctl status mosquitto

# 2. 手动测试发布/订阅
mosquitto_pub -h <broker_ip> -t "test/topic" -m "hello"
mosquitto_sub -h <broker_ip> -t "test/topic" -v

# 3. 检查防火墙
sudo ufw status
# MQTT 默认端口 1883 需要放行
```

### HTTP 设备超时

```bash
# 1. 检查设备是否在线
ping <device_ip>

# 2. 手动测试端点
curl http://<device_ip>:8080/state
curl -X POST http://<device_ip>:8080/command \
  -H "Content-Type: application/json" \
  -d '{"command":"start","params":{}}'

# 3. 检查防火墙和端口
nmap -p 8080 <device_ip>
```

### Modbus 通信失败

```bash
# 1. 检查串口权限（Linux）
ls -la /dev/ttyUSB*
sudo usermod -a -G dialout $USER  # 添加用户到 dialout 组

# 2. 使用 modpoll 测试
# 下载: https://www.modbusdriver.com/modpoll.html
modpoll -m rtu -a 1 -r 0 -c 4 -b 9600 -p none /dev/ttyUSB0

# 3. 检查接线
# A+ ↔ A+, B- ↔ B-, GND ↔ GND
```
